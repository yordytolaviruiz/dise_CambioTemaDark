Código de Ejemplo

// JavaScript para cambiar tema
const toggleTheme = () => {
    document.body.classList.toggle('dark-mode');
    localStorage.setItem('theme', 
        document.body.classList.contains('dark-mode') 
        ? 'dark' : 'light'
    );
};
